```python
def rasterize(book_id: str, pdf_path: Path) -> list[dict]:
    out_dir = DATA_RAW / book_id
    out_dir.mkdir(parents=True, exist_ok=True)
    sha256 = hashlib.sha256(pdf_path.read_bytes()).hexdigest()
    records = []
    with pymupdf.open(pdf_path) as doc:
        zoom = DPI / 72.0
        mat = pymupdf.Matrix(zoom, zoom)
        n = doc.page_count
        for i, page in enumerate(doc, start=1):
            out_path = out_dir / f"{i:04d}.png"
            if not out_path.exists():
                pix = page.get_pixmap(matrix=mat, colorspace=pymupdf.csRGB, alpha=False)
                pix.save(out_path)
            records.append({
                "book_id": book_id,
                "page_num": i,
                "image_path": str(out_path).replace("\\", "/"),
                "source_url": BOOKS[book_id]["url"],
                "pdf_sha256": sha256,
            })
    return records
```

```python
_HEADING_FONT_PT = 13.0
_MERGE_GAP_PT = 11.0
_COLUMN_BUCKET_PT = 60.0

_source_pdf_cache: dict[str, pymupdf.Document] = {}

def _open_source_pdf(doc_id: str) -> pymupdf.Document:
    if doc_id not in _source_pdf_cache:
        path = Path("data/raw/_source_pdfs") / f"{doc_id}.pdf"

def _scale_rect(rect: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    x0, y0, x1, y1 = rect
    return (int(x0 * _PDF_SCALE), int(y0 * _PDF_SCALE), int(x1 * _PDF_SCALE), int(y1 * _PDF_SCALE))
```

Now for the merge function - this is important because PyMuPDF splits text into fine slivers, and Nougat needs larger regions for context:

```python
def _merge_text_blocks(
    blocks: list[tuple[tuple[float, float, float, float], str]],
) -> list[tuple[tuple[float, float, float, float], str]]:
    by_col: dict[int, list] = {}
    for bbox, kind in blocks:
        col = round(bbox[0] / _COLUMN_BUCKET_PT)
        by_col.setdefault(col, []).append((bbox, kind))
    
    merged = []
    for col_blocks in by_col.values():
        col_blocks.sort(key=lambda b: b[0][1])
        cur = None
        cur_kind = "text"
        for bbox, kind in col_blocks:
            if cur is not None and kind == cur_kind and bbox[1] - cur[3] <= _MERGE_GAP_PT:
                cur[0], cur[1] = min(cur[0], bbox[0]), min(cur[1], bbox[1])
                cur[2], cur[3] = max(cur[2], bbox[2]), max(cur[3], bbox[3])
            else:
                if cur is not None:
                    merged.append(((cur[0], cur[1], cur[2], cur[3]), cur_kind))
                cur, cur_kind = list(bbox), kind
        if cur is not None:
            merged.append(((cur[0], cur[1], cur[2], cur[3]), cur_kind))
    return merged
```

**AI:** The column bucketing prevents multi-column exercise sets from being merged across columns. The merge gap of 11pt is about one line of text, which is appropriate for these textbooks.

```python
class Reader:
    def __init__(self, cfg: dict) -> None:
        self.cfg = cfg["ocr"]
        self._model = None
        self._processor = None
        self._device = "cpu"

    def _load(self) -> None:
        if self._model is not None:
            return
        import torch
        from transformers import NougatProcessor, VisionEncoderDecoderModel

        self._device = "cuda" if torch.cuda.is_available() else "cpu"
```

**Me:** For the crop function, I need to pad regions so strokes at the edge aren't clipped:

```python
def _crop(self, im: Image.Image, bbox: tuple[int, int, int, int]) -> Image.Image:
    x0, y0, x1, y1 = bbox
    if x1 <= x0 or y1 <= y0:
        return im  # degenerate bbox -> use full page
    w, h = x1 - x0, y1 - y0
    pad_x, pad_y = int(w * _CROP_PAD_FRAC), int(h * _CROP_PAD_FRAC)
    left, top = max(0, x0 - pad_x), max(0, y0 - pad_y)
    right, bottom = min(im.width, x1 + pad_x), min(im.height, y1 + pad_y)
    return im.crop((left, top, right, bottom))
```

And the transcription:

```python
def transcribe_region(self, region: Region) -> str:
    self._load()
    import torch
    
    image_path = image_path_for(region.page_id)
    with Image.open(image_path) as raw:
        crop = self._crop(raw.convert("RGB"), region.bbox)
    
    pixel_values = self._processor(crop, return_tensors="pt").pixel_values.to(self._device)
    with torch.no_grad():
        outputs = self._model.generate(
            pixel_values,
            min_length=1,
            max_new_tokens=_MAX_NEW_TOKENS,
            bad_words_ids=[[self._processor.tokenizer.unk_token_id]],
        )
    sequence = self._processor.batch_decode(outputs, skip_special_tokens=True)[0]
    sequence = self._processor.post_process_generation(sequence, fix_markdown=False)
    return sequence.strip()
```

**AI:** The 2% padding prevents superscripts/subscripts from being clipped. The `bad_words_ids` prevents unknown tokens. `fix_markdown=False` keeps the LaTeX intact.
